\textbf{Ricerca del massimo}
\begin{unipdcode}[language=Python,
  basicstyle=\ttfamily\scriptsize,
  numbers=none,
  xleftmargin=0pt,
  xrightmargin=0pt,
  framexleftmargin=0.3em]
def find_max(v, v_size):
    if v_size == 0:
        raise ValueError("Lista vuota")
    massimo = v[0]
    for i in range(1, v_size):
        if v[i] > massimo:
            massimo = v[i]
    return massimo
\end{unipdcode}
