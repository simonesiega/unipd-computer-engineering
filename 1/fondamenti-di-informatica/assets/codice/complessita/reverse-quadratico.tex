def reverse_quadratico(lista):
    n = len(lista)                               # 1
    output = [0] * n                             # n + 1
    output_size = 0                              # 1
    for i in range(n):                           # n
        for j in range(output_size, 0, -1):      # i
            output[j] = output[j - 1]            # 3
        output[0] = lista[i]                     # 2
        output_size += 1                         # 1
    return output                                # escluso
