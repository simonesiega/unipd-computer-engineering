public static int[] remove(
        int[] v, int index) {
    if (v == null || index < 0
            || index >= v.length)
        throw new IllegalArgumentException();
    int[] out = resize(v, v.length - 1);
    if (index < out.length)
        out[index] = v[v.length - 1];
    return out;
}
