public static String printArray(int[] v) {
    if (v == null) throw new IllegalArgumentException();
    String testo = "[";
    for (int i = 0; i < v.length; i++) {
        if (i > 0) testo += ", ";
        testo += v[i];
    }
    return testo + "]";
}
