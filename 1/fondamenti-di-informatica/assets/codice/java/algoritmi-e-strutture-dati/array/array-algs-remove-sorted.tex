public static int[] removeSorted(
        int[] v, int index) {
    if (v == null || index < 0
            || index >= v.length)
        throw new IllegalArgumentException();
    int[] out = new int[v.length - 1];
    for (int i = 0; i < index; i++)
        out[i] = v[i];
    for (int i = index; i < out.length; i++)
        out[i] = v[i + 1];
    return out;
}
