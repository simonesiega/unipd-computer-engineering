public static int binarySearch(int[] v, int value) {
    if (v == null) throw new IllegalArgumentException();
    int left = 0, right = v.length - 1;
    while (left <= right) {
        int mid = left + (right - left) / 2;
        if (v[mid] == value) return mid;
        if (v[mid] < value) left = mid + 1;
        else right = mid - 1;
    }
    return -1;
}
